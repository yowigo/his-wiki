
CREATE TABLE public.pt_balance (
	id int8 NOT NULL, -- id
	org_id varchar(36) NOT NULL, -- 机构id
	rec_prop int2 NOT NULL, -- 记录性质
	rec_status int2 NOT NULL, -- 记录状态
	"no" varchar(20) NOT NULL, -- 结帐单号
	pt_id varchar(36) NOT NULL, -- 患者id
	pt_name varchar(100) NULL, -- 姓名
	pt_sex varchar(2) NULL, -- 性别编码
	pt_age varchar(20) NULL, -- 年龄
	pt_birthday timestamp NULL, -- 出生日期
	id_card_no varchar(25) NULL, -- 身份证号
	business_no varchar(20) NULL, -- 业务号码:门诊号;住院号;体检号;门特号
	bed_no varchar(20) NULL, -- 床号
	invoice_no json NULL, -- 发票号
	invoice_qty int2 NULL, -- 发票数
	fee_start_date timestamp(6) NULL, -- 费用开始日期
	fee_end_date timestamp(6) NULL, -- 费用截止日期
	inp_start_date timestamp(6) NULL, -- 住院开始日期
	inp_end_date timestamp(6) NULL, -- 住院截止日期
	operator_id varchar(36) NULL, -- 操作员id
	operator_name varchar(60) NULL, -- 操作员名
	operate_dept varchar(100) NULL, -- 操作部门名称
	operate_dept_id varchar(36) NULL, -- 操作部门id
	balance_time timestamp(6) NULL, -- 结算时间
	visit_id varchar(36) NULL, -- 就诊id
	print_times int2 NULL, -- 打印次数
	data_source varchar(30) NULL, -- 结算数据来源:医院窗口/诊间支付/公众号/自助机/……
	invoice_source int2 NULL, -- 发票来源
	complement_prop int2 NULL, -- 补结性质:1-补充结算产生;2-已补充结算的原始记录;3-共济结算产生;4-已共济结算的原始记录
	complement_id int8 NULL, -- 补结id：当补结性质=1,存已补充结算的原始ID; 补结性质=2,存补充结算新ID; 补结性质=3,存已共济结算的原始ID; 补结性质=4,存共济结算新ID;
	ins_id int8 NULL, -- 保险系统
	ins_name varchar(60) NULL, -- 保险名称
	balance_status int2 NULL, -- 结算状态 0-正常 完成结算;1-异常 可能医保成功,本地未成功 ;2-医保部分退,全退成功,未进行重收
	link_id int8 NULL, -- 相关id
	description varchar(200) NULL, -- 说明
	module_source int2 NULL, -- 内部模块来源:1-门诊医嘱(发送产生);2-住院费用(出院时产生);3-体检费用(发送时产生);4-挂号费用(划价);5-手工划价;6-飞跃;7-外部接口;8-门特费用
	py_code varchar(100) NULL, -- 姓名拼音简码
	discount_id varchar(36) NULL, -- 折扣id
	discount_name varchar(50) NULL, -- 折扣名称
	receivable_charge numeric(18, 5) NULL, -- 应收金额
	actual_charge numeric(18, 5) NULL, -- 实收金额
	pt_sex_name varchar(20) NULL, -- 性别名称
	balance_type int2 NULL, -- 当为结账记录rec_prop=2时:结账类型:0-出院结账;1-中途结账
	refund_times int2 NULL, -- 退费次数:用于记录患者的退费次,主要是记录门诊部分退费
	balance_lot int8 NULL, -- 结算批次:用于门特循环一批结算,或后期体检批量结算
	pdr_id varchar(36) NULL, -- 责任医师id
	pdr_name varchar(60) NULL, -- 责任医师
	dept_id varchar(36) NULL, -- 科室id
	dept_name varchar(100) NULL, -- 科室名称
	insurance_type_code varchar(20) NULL, -- 医保险种类型编码, 医保结算时由医保返回并保存
	insurance_type_name varchar NULL, -- 医保险种类型名称, 医保结算时由医保返回并保存
	is_body int2 DEFAULT 0 NULL, -- 是否躯体疾病结算: 0-原始就诊结算(精神疾病) 1-拆分就诊结算(躯体疾病)
	english_name varchar(100) NULL, -- 患者英文名
	pt_tag varchar(50) NULL, -- 患者个人标签
	tel varchar(50) NULL, -- 联系电话
	invoice_type int2 NULL, -- 发票类型, 当住院结账调用门诊收费票据打印时写入
	CONSTRAINT pk_pt_balance PRIMARY KEY (id)
);
CREATE INDEX idx_pt_balance_operator ON public.pt_balance USING btree (org_id, operator_id) WHERE (operator_id IS NOT NULL);
CREATE INDEX idx_pt_balance_org_recprop ON public.pt_balance USING btree (org_id, rec_prop, id);
CREATE INDEX idx_pt_balance_rec_status ON public.pt_balance USING btree (org_id, no, rec_status);
CREATE INDEX "ix_pt_balance$balance_lot" ON public.pt_balance USING btree (balance_lot);
CREATE INDEX "ix_pt_balance$balance_time" ON public.pt_balance USING btree (balance_time, org_id);
CREATE INDEX "ix_pt_balance$business_no" ON public.pt_balance USING btree (org_id, business_no);
CREATE INDEX "ix_pt_balance$complement" ON public.pt_balance USING btree (complement_id);
CREATE INDEX "ix_pt_balance$link" ON public.pt_balance USING btree (link_id);
CREATE INDEX "ix_pt_balance$pt" ON public.pt_balance USING btree (pt_id);
CREATE INDEX "ix_pt_balance$visit" ON public.pt_balance USING btree (visit_id);
CREATE INDEX pt_balance_tel_idx ON public.pt_balance USING btree (tel);
CREATE UNIQUE INDEX uq_pt_balance ON public.pt_balance USING btree (no, rec_status, link_id, org_id);
COMMENT ON TABLE public.pt_balance IS '患者结算记录';

CREATE TABLE public.pt_balance_detail (
	id int8 NOT NULL, -- id
	balance_id int8 NOT NULL, -- 结算记录id
	balance_name varchar(50) NOT NULL, -- 结算方式
	balance_charge numeric(18, 5) NULL, -- 结算金额，正式的结算金额小数位数为2位，误差费，小数位数较大
	balance_interface varchar(50) NULL, -- 结算接口名称
	balance_interface_no varchar(70) NULL, -- 结算接口号码
	description varchar(1000) NULL, -- 说明
	is_prepay int2 NULL, -- 是否预交抵扣
	prepay_usage_id int8 NULL, -- 预交使用明细ID
	balance_prop int2 NULL, -- 结算性质
	CONSTRAINT pk_pt_balance_detail PRIMARY KEY (id),
	CONSTRAINT "fk_balance_details$preapy_usa_id" FOREIGN KEY (prepay_usage_id) REFERENCES public.pt_prepay_usage_details(id),
	CONSTRAINT "fk_pt_balance_detail$balance_id" FOREIGN KEY (balance_id) REFERENCES public.pt_balance(id) ON DELETE RESTRICT ON UPDATE RESTRICT
);
CREATE INDEX idx_pt_balance_detail_balance ON public.pt_balance_detail USING btree (balance_id);
CREATE INDEX "ix_pt_balance_detail$balance" ON public.pt_balance_detail USING btree (balance_id);
CREATE INDEX "ix_pt_balance_detail$prepay_usa" ON public.pt_balance_detail USING btree (prepay_usage_id);
CREATE UNIQUE INDEX uq_pt_balance_detail ON public.pt_balance_detail USING btree (balance_id, prepay_usage_id, balance_name);
COMMENT ON TABLE public.pt_balance_detail IS '患者结算明细';

	invoicecode varchar(20) NULL, -- 发票代码

	qrcode varchar(500) NULL, -- 二维码
	machinecode varchar(100) NULL, -- 税控设备号

	paperpdfurl varchar(500) NULL, -- 含底图纸票pdf地址
	ofdurl varchar(500) NULL, -- 发票ofd地址
	xmlurl varchar(500) NULL, -- 发票xml地址
	clerk varchar(50) NULL, -- 开票员
	payee varchar(50) NULL, -- 收款人
	checker varchar(50) NULL, -- 复核人
	saleraccount varchar(100) NULL, -- 销方银行账号

	oldinvoiceno varchar(50) NULL, -- 对应蓝票发票号码
	oldeleinvoicenumber varchar(50) NULL, -- 对应蓝票数电票号码
	listflag varchar(2) NULL, -- 清单标志
	listname varchar(50) NULL, -- 清单项目名称
	phone varchar(50) NULL, -- 购方手机
	notifyemail varchar(50) NULL, -- 购方邮箱推送邮箱
	vehicleflag varchar(2) NULL, -- 是否机动车类专票
	createtime varchar(30) NULL, -- 数据创建时间
	updatetime varchar(30) NULL, -- 数据更新时间

	proxyinvoiceflag varchar(2) NULL, -- 代开标志
	invoicedate varchar(30) NULL, -- 用于开票的订单的时间
	invoicetype varchar(2) NULL, -- 开票类型
	redreason varchar(200) NULL, -- 冲红原因

	bfield2 varchar(500) NULL, -- 业务方自定义字段2
	bfield3 varchar(500) NULL, -- 业务方自定义字段3
	naturalpersonflag varchar(50) NULL, -- 购买方自然人标志
	invoiceitems json NULL, -- 发票明细集合
	vehicleinfo json NULL, -- 机动车销售统一发票中机动车相关信息
	secondhandcarinfo json NULL, -- 二手车销售统一发票时才有值返回

	realpropertyrentinfo json NULL, -- 数电不动产经营租赁服务特定要素类型的发票

	additionalelementlist json NULL, -- 附加要素信息列表
